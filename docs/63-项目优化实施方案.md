# backend/app/services/chat/message_saver.py
"""
消息保存服务

统一处理消息和记忆的保存逻辑，避免代码重复
"""
from typing import Optional
from sqlalchemy.orm import Session
from app.engines.memory.manager import MemoryManager
from app.core.personality.models import Personality
from app.models.message import Message as MessageModel
from app.models.session import Session as SessionModel
from app.utils.logger import logger
import uuid
from datetime import datetime


class MessageSaver:
    """消息保存服务
    
    统一处理消息和记忆的保存，支持流式和非流式两种模式
    """
    
    def __init__(
        self,
        db: Session,
        memory_manager: MemoryManager
    ):
        self.db = db
        self.memory_manager = memory_manager
    
    async def save_conversation_turn(
        self,
        user_id: str,
        session_id: str,
        user_message: str,
        assistant_message: str,
        assistant_model: str,
        personality: Optional[Personality] = None,
        use_memory: bool = True
    ) -> dict:
        """保存一轮对话（用户消息+助手消息）
        
        Args:
            user_id: 用户ID
            session_id: 会话ID
            user_message: 用户消息内容
            assistant_message: 助手消息内容
            assistant_model: 使用的模型名称
            personality: 人格配置
            use_memory: 是否保存到记忆系统
            
        Returns:
            dict: 保存结果
        """
        try:
            session_uuid = uuid.UUID(session_id)
            user_uuid = uuid.UUID(user_id)
            
            # 1. 保存用户消息到数据库
            user_msg = MessageModel(
                session_id=session_uuid,
                user_id=user_uuid,
                role="user",
                content=user_message
            )
            self.db.add(user_msg)
            
            # 2. 保存助手消息到数据库
            assistant_msg = MessageModel(
                session_id=session_uuid,
                user_id=user_uuid,
                role="assistant",
                content=assistant_message,
                model=assistant_model
            )
            self.db.add(assistant_msg)
            
            # 3. 更新会话统计
            session = self.db.query(SessionModel).filter(
                SessionModel.id == session_uuid
            ).first()
            if session:
                session.message_count = (session.message_count or 0) + 2  # type: ignore[assignment]
                session.last_message_at = datetime.utcnow()  # type: ignore[assignment]
            
            self.db.commit()
            
            logger.debug(
                "Messages saved to database",
                extra={
                    "user_id": user_id,
                    "session_id": session_id,
                    "user_message_length": len(user_message),
                    "assistant_message_length": len(assistant_message)
                }
            )
            
            # 4. 保存到记忆系统（如果启用）
            if personality and personality.memory.enabled and use_memory:
                await self.memory_manager.add_conversation_turn(
                    user_id=user_id,
                    session_id=session_id,
                    user_message=user_message,
                    assistant_message=assistant_message,
                    async_save=True
                )
                
                logger.debug(
                    "Conversation saved to memory",
                    extra={
                        "user_id": user_id,
                        "session_id": session_id
                    }
                )
            
            return {
                "success": True,
                "user_message_id": str(user_msg.id),
                "assistant_message_id": str(assistant_msg.id)
            }
            
        except Exception as e:
            self.db.rollback()
            logger.error(
                f"Failed to save conversation turn: {e}",
                exc_info=True,
                extra={
                    "user_id": user_id,
                    "session_id": session_id
                }
            )
            return {
                "success": False,
                "error": str(e)
            }

# backend/app/services/chat/tool_handler.py
"""
工具调用处理器

处理工具调用的完整流程：解析、执行、结果处理
"""
from typing import List, Dict, Any, Optional
from app.engines.tools.factory import ToolManagerFactory
from app.engines.ai.base import ChatMessage as EngineChatMessage
from app.utils.logger import logger
import json


class ToolCallHandler:
    """工具调用处理器"""
    
    def __init__(self, tool_factory: ToolManagerFactory):
        self.tool_factory = tool_factory
    
    async def handle_tool_calls(
        self,
        tool_call_chunks: List[Dict[str, Any]],
        current_messages: List[EngineChatMessage],
        max_tokens: Optional[int] = None
    ) -> tuple[List[EngineChatMessage], List[Dict[str, Any]]]:
        """处理工具调用
        
        Args:
            tool_call_chunks: 工具调用列表
            current_messages: 当前消息列表
            max_tokens: 最大token数（用于截断过长结果）
            
        Returns:
            tuple: (更新后的消息列表, 工具执行结果列表)
        """
        tool_manager = self.tool_factory.get_tool_manager()
        tool_results = []
        
        for tc in tool_call_chunks:
            if not tc or not tc.get("function", {}).get("name"):
                continue
            
            tool_name = tc["function"]["name"]
            
            try:
                # 1. 解析参数
                args = json.loads(tc["function"].get("arguments", "{}"))
                
                # 2. 执行工具
                result = await tool_manager.execute_tool(
                    tool_name=tool_name,
                    parameters=args
                )
                
                # 3. 格式化结果
                if result.get("success"):
                    result_content = result.get("result", "")
                    
                    # 转换为字符串
                    if isinstance(result_content, (dict, list)):
                        result_content = json.dumps(result_content, ensure_ascii=False)
                    else:
                        result_content = str(result_content)
                    
                    # 4. 截断过长的结果
                    if max_tokens:
                        result_content = self._truncate_result(
                            result_content,
                            max_tokens,
                            tool_name
                        )
                    
                    tool_results.append({
                        "role": "tool",
                        "tool_call_id": tc.get("id", ""),
                        "content": result_content
                    })
                else:
                    tool_results.append({
                        "role": "tool",
                        "tool_call_id": tc.get("id", ""),
                        "content": f"工具执行失败: {result.get('error', 'Unknown error')}"
                    })
                
                logger.info(
                    f"Tool executed: {tool_name}",
                    extra={"tool_name": tool_name, "success": result.get("success")}
                )
                
            except Exception as e:
                logger.error(f"Tool execution error: {e}", exc_info=True)
                tool_results.append({
                    "role": "tool",
                    "tool_call_id": tc.get("id", ""),
                    "content": f"工具执行出错: {str(e)}"
                })
        
        # 5. 更新消息列表
        if tool_results:
            # 添加assistant消息（包含tool_calls）
            assistant_tool_calls = [
                {
                    "id": tc.get("id", ""),
                    "type": tc.get("type", "function"),
                    "function": tc.get("function", {})
                }
                for tc in tool_call_chunks
                if tc and tc.get("function", {}).get("name")
            ]
            
            if assistant_tool_calls:
                current_messages.append(
                    EngineChatMessage(
                        role="assistant",
                        content="",
                        tool_calls=assistant_tool_calls
                    )
                )
            
            # 添加tool消息
            for tr in tool_results:
                current_messages.append(
                    EngineChatMessage(
                        role="tool",
                        content=tr.get("content", ""),
                        name=tr.get("tool_call_id", "")
                    )
                )
        
        return current_messages, tool_results
    
    def _truncate_result(
        self,
        content: str,
        max_tokens: int,
        tool_name: str
    ) -> str:
        """截断过长的工具结果
        
        Args:
            content: 原始内容
            max_tokens: 最大token数
            tool_name: 工具名称
            
        Returns:
            str: 截断后的内容
        """
        from app.utils.token_utils import estimate_tokens
        
        # 预留20%的token给工具结果
        available_tokens = int(max_tokens * 0.2) if max_tokens else 500
        result_tokens = estimate_tokens(content)
        
        if result_tokens > available_tokens:
            # 计算可保留的字符数（约2字符/token）
            max_chars = available_tokens * 2
            if len(content) > max_chars:
                truncated = content[:max_chars] + f"\n\n[结果已截断，原始长度: {len(content)} 字符]"
                
                logger.warning(
                    f"Tool result truncated: {len(content)} -> {len(truncated)} chars",
                    extra={
                        "tool_name": tool_name,
                        "original_length": len(content),
                        "truncated_length": len(truncated),
                        "original_tokens": result_tokens,
                        "truncated_tokens": estimate_tokens(truncated)
                    }
                )
                
                return truncated
        
        return content

# backend/app/services/memory/scoring_service.py
"""
记忆评分服务

实现多因子评分算法，提升记忆检索质量
"""
from typing import List, Dict, Any, Optional
from app.core.personality.models import IntelligentScoring, ScoringWeights
from app.engines.memory.models import MemorySearchResult
from app.utils.logger import logger
import re


class MemoryScoringService:
    """记忆评分服务"""
    
    def apply_intelligent_scoring(
        self,
        memories: List[MemorySearchResult],
        query: str,
        config: IntelligentScoring
    ) -> List[MemorySearchResult]:
        """应用智能评分
        
        Args:
            memories: 记忆列表
            query: 查询文本
            config: 智能评分配置
            
        Returns:
            List[MemorySearchResult]: 重新排序后的记忆列表
        """
        # 检测查询类型
        is_numeric_query = any(
            keyword in query 
            for keyword in config.numeric_query_keywords
        )
        
        # 计算评分
        scored_memories = []
        for mem in memories:
            score = self._calculate_score(mem, query, config, is_numeric_query)
            scored_memories.append((score, mem))
        
        # 排序
        scored_memories.sort(key=lambda x: x[0], reverse=True)
        
        # 记录日志
        logger.debug(
            "Applied intelligent scoring",
            extra={
                "query": query[:50],
                "is_numeric_query": is_numeric_query,
                "top_5_scores": [
                    {"content": m.memory.content[:50], "score": round(s, 4)}
                    for s, m in scored_memories[:5]
                ]
            }
        )
        
        return [m for _, m in scored_memories]
    
    def _calculate_score(
        self,
        mem_result: MemorySearchResult,
        query: str,
        config: IntelligentScoring,
        is_numeric_query: bool
    ) -> float:
        """计算记忆的智能评分"""
        content = mem_result.memory.content
        similarity = mem_result.similarity
        
        # 基础分数：相似度
        score = similarity
        
        # 数值查询加分
        if is_numeric_query:
            has_number = bool(re.search(r'\d+', content))
            if has_number:
                score += config.numeric_boost
                
                # 关键词匹配加分
                if any(kw in content for kw in config.numeric_memory_keywords):
                    score += config.keyword_match_boost
        
        # 内容长度加分
        content_len = len(content)
        if config.optimal_length_min <= content_len <= config.optimal_length_max:
            score += config.length_boost
        
        return score
    
    def apply_comprehensive_scoring(
        self,
        memories: List[MemorySearchResult],
        query: str,
        session_id: str,
        weights: ScoringWeights,
        memory_levels_config: Optional[Any] = None
    ) -> List[MemorySearchResult]:
        """应用多因子综合评分
        
        考虑相似度、重要性、时效性、相关性等多个因素
        """
        scored_memories = []
        for mem in memories:
            score = self._calculate_comprehensive_score(
                mem,
                query,
                session_id,
                weights,
                memory_levels_config
            )
            scored_memories.append((score, mem))
        
        scored_memories.sort(key=lambda x: x[0], reverse=True)
        return [m for _, m in scored_memories]
    
    def _calculate_comprehensive_score(
        self,
        mem_result: MemorySearchResult,
        query: str,
        session_id: str,
        weights: ScoringWeights,
        memory_levels_config: Optional[Any]
    ) -> float:
        """计算综合评分"""
        memory = mem_result.memory
        
        # 1. 相似度分数
        similarity_score = mem_result.similarity
        
        # 2. 重要性分数
        importance_score = memory.importance
        
        # 3. 时效性分数
        recency_score = self._calculate_recency_score(memory.created_at)
        
        # 4. 相关性分数
        relevance_score = self._calculate_relevance_score(memory.content, query)
        
        # 5. 加权求和
        comprehensive_score = (
            similarity_score * weights.similarity +
            importance_score * weights.importance +
            recency_score * weights.recency +
            relevance_score * weights.relevance
        )
        
        # 6. 记忆层级加权
        if memory_levels_config and memory_levels_config.enabled:
            if memory.session_id == session_id:
                comprehensive_score *= memory_levels_config.session_memory_boost
            else:
                comprehensive_score *= memory_levels_config.cross_session_memory_boost
        
        return comprehensive_score
    
    def _calculate_recency_score(self, created_at) -> float:
        """计算时效性分数"""
        from datetime import datetime, timezone
        
        if not created_at:
            return 0.5
        
        if isinstance(created_at, datetime):
            now = datetime.now(timezone.utc) if created_at.tzinfo else datetime.now()
            delta = (now - created_at).total_seconds() / 86400  # 天数
        else:
            return 0.5
        
        # 指数衰减
        if delta <= 7:
            return 1.0
        elif delta <= 30:
            return 0.5 + 0.5 * (1 - (delta - 7) / 23)
        elif delta <= 90:
            return 0.1 + 0.4 * (1 - (delta - 30) / 60)
        else:
            return 0.1
    
    def _calculate_relevance_score(self, content: str, query: str) -> float:
        """计算相关性分数"""
        query_words = set(re.findall(r'\w+', query.lower()))
        content_words = set(re.findall(r'\w+', content.lower()))
        
        if query_words:
            jaccard = len(query_words & content_words) / len(query_words | content_words)
        else:
            jaccard = 0.0
        
        return min(1.0, jaccard)
```

"""提示词加载器"""
import yaml
from pathlib import Path
from typing import Dict, Any, Optional
from jinja2 import Template
from app.utils.logger import logger


class PromptLoader:
    """提示词加载器"""
    
    def __init__(self, prompts_dir: Optional[str] = None):
        if prompts_dir is None:
            base_dir = Path(__file__).parent.parent.parent.parent
            prompts_dir = str(base_dir / "config" / "prompts")
        
        self.prompts_dir = Path(prompts_dir)
        self.cache: Dict[str, Dict[str, Any]] = {}
    
    def load_response_style(self, style: str) -> Optional[Dict[str, Any]]:
        """加载回答风格模板"""
        cache_key = f"response_styles.{style}"
        
        if cache_key in self.cache:
            return self.cache[cache_key]
        
        config_path = self.prompts_dir / "response_styles" / f"{style}.yaml"
        
        if not config_path.exists():
            logger.warning(f"Response style not found: {style}")
            return None
        
        with open(config_path, 'r', encoding='utf-8') as f:
            config = yaml.safe_load(f)
        
        self.cache[cache_key] = config
        return config
    
    def render_template(self, template_str: str, variables: Dict[str, Any]) -> str:
        """渲染模板"""
        template = Template(template_str)
        return template.render(**variables)
    
    def clear_cache(self):
        """清除缓存（用于热更新）"""
        self.cache.clear()


# 全局实例
_prompt_loader: Optional[PromptLoader] = None

def get_prompt_loader() -> PromptLoader:
    global _prompt_loader
    if _prompt_loader is None:
        _prompt_loader = PromptLoader()
    return _prompt_loader

        self,

        code: str = "ERROR",

        self.code = code
        self.details = details or {}
        super().__init__(message)
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典格式"""
        return {

            "message": self.message,
            "details": self.details

async def cozy_exception_handler(request: Request, exc: CozyError) -> JSONResponse:
    """处理CozyError异常"""
    
    # 确定HTTP状态码
    if isinstance(exc, AuthenticationError):
        status_code = status.HTTP_401_UNAUTHORIZED
    elif isinstance(exc, AuthorizationError):
        status_code = status.HTTP_403_FORBIDDEN
    elif isinstance(exc, ResourceNotFoundError):
        status_code = status.HTTP_404_NOT_FOUND
    elif isinstance(exc, ValidationError):
        status_code = status.HTTP_400_BAD_REQUEST
    elif isinstance(exc, BusinessLogicError):
        status_code = status.HTTP_422_UNPROCESSABLE_ENTITY
    elif isinstance(exc, ExternalServiceError):
        status_code = status.HTTP_503_SERVICE_UNAVAILABLE
    else:
        status_code = status.HTTP_500_INTERNAL_SERVER_ERROR
    
    # 记录日志
    logger.error(
        f"CozyError: {exc.code}",
        exc_info=True,
        extra={
            "error_code": exc.code,
            "error_message": exc.message,
            "error_details": exc.details,
            "path": request.url.path,
            "method": request.method
        }
    )
    
    # 返回标准错误响应
    return JSONResponse(
        status_code=status_code,
        content=exc.to_dict()
    )

    """引擎基类
    
    所有引擎的抽象基类，提供公共功能
    """
    
    def __init__(self, config: Dict[str, Any]):
        self.config = config
        self._validate_config()
        logger.info(
            f"{self.__class__.__name__} initialized",
            extra={"config": self._safe_config_for_logging()}
        )
    
    @abstractmethod
    def _validate_config(self) -> None:
        """验证配置（子类实现）"""
        pass
    
    def _safe_config_for_logging(self) -> Dict[str, Any]:
        """返回安全的配置（隐藏敏感信息）"""
        safe_config = self.config.copy()
        
        # 隐藏敏感字段
        sensitive_fields = ["api_key", "secret_key", "password", "token"]
        for field in sensitive_fields:
            if field in safe_config:
                safe_config[field] = "***"
        
        return safe_config
    
    def _handle_error(self, error: Exception, context: Optional[Dict[str, Any]] = None):
        """统一的错误处理"""
        logger.error(
            f"{self.__class__.__name__} error: {error}",
            exc_info=True,
            extra={"context": context or {}}
        )
        raise
```

使用joinedload避免N+1查询

from sqlalchemy.orm import joinedload
from sqlalchemy import select


@router.get("/", response_model=List[SessionResponse])
async def list_sessions(
    db: Session = Depends(get_sync_session),
    user: User = Depends(get_current_active_user),
    limit: int = Query(20, ge=1, le=100),
    offset: int = Query(0, ge=0)
):
    """列出用户的所有会话（优化版）"""
    
    # 使用joinedload预加载关联数据，避免N+1查询
    stmt = (
        select(SessionModel)
        .where(
            and_(
                SessionModel.user_id == user.id,
                SessionModel.deleted_at.is_(None)
            )
        )
        .options(
            joinedload(SessionModel.messages).joinedload(MessageModel.user)
        )
        .order_by(SessionModel.updated_at.desc())
        .limit(limit)
        .offset(offset)
    )
    
    result = db.execute(stmt)
    sessions = result.unique().scalars().all()
    
    return sessions

# scripts/sync_docs.py (改进)
"""
文档同步脚本（改进版）

自动检查代码变更，提示需要更新的文档
"""
import os
import re
from pathlib import Path
from typing import List, Dict, Set
import git


class DocSyncChecker:
    """文档同步检查器"""
    
    def __init__(self, repo_path: str = "."):
        self.repo = git.Repo(repo_path)
        self.changed_files: Set[str] = set()
        self.affected_docs: Dict[str, List[str]] = {}
    
    def check_recent_changes(self, commits: int = 10):
        """检查最近的代码变更"""
        recent_commits = list(self.repo.iter_commits(max_count=commits))
        
        for commit in recent_commits:
            for item in commit.stats.files:
                if item.endswith('.py') or item.endswith('.ts'):
                    self.changed_files.add(item)
        
        print(f"检查到 {len(self.changed_files)} 个文件变更")
    
    def find_affected_docs(self):
        """查找受影响的文档"""
        doc_mappings = {
            "backend/app/api/v1/": ["docs/04-API接口设计.md"],
            "backend/app/models/": ["docs/05-数据库设计.md"],
            "backend/app/core/": ["docs/02-后端架构设计.md"],
            "frontend/src/": ["docs/03-前端架构设计.md"],
        }
        
        for file in self.changed_files:
            for path_prefix, docs in doc_mappings.items():
                if file.startswith(path_prefix):
                    if file not in self.affected_docs:
                        self.affected_docs[file] = []
                    self.affected_docs[file].extend(docs)
        
        return self.affected_docs
    
    def generate_report(self):
        """生成同步报告"""
        print("\n===== 文档同步检查报告 =====\n")
        
        if not self.affected_docs:
            print("✅ 没有检测到需要更新的文档")
            return
        
        print("⚠️ 以下文档可能需要更新：\n")
        
        unique_docs = set()
        for docs in self.affected_docs.values():
            unique_docs.update(docs)
        
        for doc in sorted(unique_docs):
            affected_files = [
                f for f, docs in self.affected_docs.items()
                if doc in docs
            ]
            print(f"📄 {doc}")
            print(f"   受影响的文件：")
            for f in affected_files[:5]:  # 只显示前5个
                print(f"   - {f}")
            if len(affected_files) > 5:
                print(f"   ... 还有 {len(affected_files) - 5} 个文件")
            print()


if __name__ == "__main__":
    checker = DocSyncChecker()
    checker.check_recent_changes(commits=20)
    checker.find_affected_docs()
    checker.generate_report()
