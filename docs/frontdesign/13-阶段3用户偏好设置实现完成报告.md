### 4.2 设置页面实现

```typescript
// 状态管理
const [voiceInputMode, setVoiceInputMode] = useState<'press' | 'click' | 'auto'>('auto');

// 初始化
useEffect(() => {
  if (preferences) {
    const voiceInputModeValue = preferences.voice_input_mode || 'auto';
    setVoiceInputMode(voiceInputModeValue);
  }
}, [preferences]);

// 更新处理
const handleVoiceInputModeChange = async (value: 'press' | 'click' | 'auto') => {
  const oldValue = voiceInputMode;
  setVoiceInputMode(value);
  try {
    const updated = await updateMutation.mutateAsync({ voice_input_mode: value });
    if (updated?.voice_input_mode !== undefined) {
      setVoiceInputMode(updated.voice_input_mode);
    }
  } catch (error) {
    setVoiceInputMode(oldValue); // 恢复原状态
  }
};
```

### 4.3 交互模式选择逻辑

```typescript
const getVoiceInputMode = useCallback((): 'press' | 'click' => {
  // 1. 优先使用用户偏好
  const userPreference = preferences?.voice_input_mode;
  
  if (userPreference === 'press') {
    return 'press';
  }
  if (userPreference === 'click') {
    return 'click';
  }
  
  // 2. 自动模式或未设置：根据设备类型
  return isMobile ? 'press' : 'click';
}, [preferences?.voice_input_mode, isMobile]);

const usePressMode = getVoiceInputMode() === 'press';
```
